1. xlwings 사용 => 열린 엑셀 파일을 사용할 때

import xlwings as wx

# 현재 열린 엑셀 파일의 활성화 된 sheet의 데이터 가져옴
wb = xw.apps.active.books.active
sheet = wb.sheets.active
print('데이터 가져와 연산 수행')

# B1과 B2의 값을 가져옵니다.
b1 = sheet.range('B1').value
b2 = sheet.range('B2').value
print(b1,b2)
# B1에서 B2를 뺀 값을 계산합니다.
result = b1 - b2
sheet.range('B3').value = result
print('연산 결과 쓰기 완료')

2. xlwings 사용 => 닫힌 파일을 열어서 처리하기

import xlwings as xw
import os
# 1. 파일 경로 설정(현재)
file_name = '1-1_xlwings_test.xlsx'
file_path = os.path.join(os.getcwd(), file_name)

print(file_path)

# 2. 엑셀 열기
wb = xw.Book(file_path)

# 3. sheet 선택
sheet = wb.sheets.active

# 4. B1과 B2의 값 연산
b1 = sheet.range('B1').value
b2 = sheet.range('B2').value
# 두 값의 차이 B3 셀에 기입
result = b1 - b2
sheet.range('B3').value = result
print('연산 결과 쓰기 완료')

# 5. 저장 및 닫기
wb.save()
wb.close()
