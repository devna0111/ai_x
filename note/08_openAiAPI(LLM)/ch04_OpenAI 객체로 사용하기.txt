from openai import OpenAI
from dotenv import load_env()

client = OpenAI()
response = client.chat.completion.create(
			model = '모델 설정',
			message = [ {'role':''system', 'content' : '당신은 ____ 페르소나 부여},
					  {'role':'user', 'content' : '유저메세지. 질문 등'},       
					  {'role':'assistant', 'content' : '대표 답변, 예씨 답변, fotmat'}, 
					  {'role':'user', 'content' : '유저메세지. 질문 등'},       
					  {'role':'assistant', 'content' : '대표 답변, 예씨 답변, fotmat'},
					  {'role':'user', 'content' : '최종 질문질문 등'},
			temperature=0.7, # 0 ~ 2 : 수가 작을 수록 message 속 표현 위주로, 수가 클 수록 창의적인 표현
		  	frequency_penalty=0.5, # 동일한 언어를 사용하거나 하지 않거나
			
			 ]
response

## 너무 답변이 길어진다, 즉 대기 시간이 길어지면 끊어내기 위해서 아래와 같이 설정하기도 함
try :
    response = client.chat.completions.create(
            model = 'gpt-4.1-nano',
            messages = [{'role':'user','content':'에러를 일으켜 볼 수 있을까요? '}],
            timeout=0.0001, # 입력 시간보다 오랜 시간동안 답변이 돌아오지 않으면 error 강제 발생
            )
except :
    print('요청시간 초과')
