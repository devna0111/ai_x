ch05_OpenAI 객체로 이미지 생성하기

챗봇은, 채팅은
client.chat.completion.create() 명령어를 사용했다면

이미지 제너레이션은
client.images.generate()로 사용

prompt_text = "숲이 보이는 밝은 자연 이미지를 그려줘"
response = client.images.generate(
    prompt = prompt_text, # 이미지 설명
    model = 'dall-e-2', # model명
    n=1, # 생성될 이미지 갯수
    size = '512x512', # 이미지 해상도 -> docs 확인해서 가능한 데이터 적용하기
)

response # ImagesResponse 객체 반환.
# 아래와 같은 코드로 url만 체크 가능
image_url = response.data[0].url

# url을 아래와 같이 PIL 라이브러리의 Imgae 객체로 열기
from PIL import Image
import requests
resp = requests.get(image_url, stream = True).raw
img = Image.open(resp)
img

# 저장은 2가지 방식

img_resp = requests.get(image_url)
img_resp.status_code

if img_resp.status_code==200 :
    with open('data/ch04_generated_dalle3.png','wb') as f :
        f.write(img_resp.content)
    print('이미지 다운로드 성공')
else :
    print('이미지 다운로드 실패')
