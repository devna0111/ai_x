#1 문서를 쪼개면서 읽기
%%time
import time
start = time.time() # currentsecond
from langchain_community.document_loaders import Docx2txtLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
loader = Docx2txtLoader('./tax_docs/소득세법(법률)(제20615호)(20250701).docx')
text_splitter = RecursiveCharacterTextSplitter( # 문서를 문자 단위(수)로 쪼갬
    chunk_size = 1500, # 문서를 쪼갤 때, 1500 토큰이 아니라 1500 글자씩 쪼개기
    chunk_overlap=200, # 200글자씩 겹치게 만들기
    
)
# 1번째 chunk : 0 ~ 1500글자
# 2번째 chunk : 1300 ~ 2800 글자
# 3번째 chunk : 2600 ~ 4100 글자.... 
document = loader.load_and_split(text_splitter)
end = time.time()
runtime = end - start
print("문서를 나누는 데 걸린 시간은",runtime,"초 입니다.")
