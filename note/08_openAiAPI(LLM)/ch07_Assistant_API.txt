# # 아래는 학습을 위해서 모든 과정을 변수에 담아서 체크. 실제로는 thread와 assistant(페르소나 부여용) 만 있어도 가능(각각의 *.id가 중요)
( client.beta.threads.create() => client.beta.threads.messages.create(...): => client.beta.threads.runs.create_and_poll(...)
# 1. 클라이언트 생성
client = OpenAI()

# 2. 어시스턴트 생성
assistant = client.beta.assistants.create(
    name = "HelpBot",
    instructions = "당신은 유능하고, 친절한 도움말 어시스턴트입니다. 사용자 질문에 20글자 이내로 친절하게 답변하세요",
    model = 'gpt-4o-mini',
#     tools 매개변수를 생략하거나 빈 리스트로 두면 기본 대화만 수행(저장))

# 3. 새로운 대화 쓰레드 생성 : 기억을 담당
thread = client.beta.threads.create()
# print('llm에 요청할 때 필요한 thread.id :',thread.id)

# 4. thread에 사용자 prompt 추가
user_message = client.beta.threads.messages.create(
    thread_id = thread.id,
    role = 'user',
    content = "집에서 키우는 생후 5주차 고양이는 사람이 없을 때 보통 뭘 할까?", # 사용자 프롬프트)

# 5. llm 요청(어시스턴트에게 답변 생성 요청) -> 답변 내용이 thread에 자동 추가
run = client.beta.threads.runs.create_and_poll(
    thread_id=thread.id,
    assistant_id = assistant.id)

print(run)

# 6. 실행 완료 후 스레드의 모든 메세지 불러오기
messages = client.beta.threads.messages.list(thread_id=thread.id)
print(messages)
# 6-1. 메세지를 시간순서대로 정렬
sorted_messages = sorted(messages.data,key = lambda x : (x.created_at, x.role)) # 객체니까 *.attr로 접근
sorted_messages

# 7. thread에 저장된 메세지들을 출력
import time
for msg in sorted_messages :
    dateStyle = time.localtime(msg.created_at)
    dateStyle_str = time.strftime('%Y-%m-%d %H:%M:%S',dateStyle)
    print("{:^9}({}) : {}".format(msg.role, dateStyle_str ,msg.content[0].text.value))
