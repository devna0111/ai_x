# 의사결정나무 : DecisionTreeClassifier를 통한 모형
from sklearn.tree import DecisionTreeClassifier
dt_model = DecisionTreeClassifier(random_state=1)
dt_model.fit(train_X, train_y)

# 위 내용 모델 생성 끝

dt_model.score(test_X,test_y) # accuracy
