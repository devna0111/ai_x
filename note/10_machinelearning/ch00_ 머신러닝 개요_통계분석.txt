# 통계요약 : 중심위치 측도(평균, 중위값, 최빈값) 및 산포 측도 체크하기
import numpy as np
from scipy.stats import kurtosis, skew

data = np.random.normal(loc = 0, scale = 1.0, size=10000000 )

# 분포도 (histogram)
import matplotlib.pyplot as plt
plt.figure(figsize=(5,1))
plt.hist(data, bins=150)
plt.show()

print('평균:',  np.mean(data))
print('중앙값:' , np.median(data))
print('분산:' , np.var(data))
print('표준편차:' , np.std(data))

# 왜도(skewness) : 0 대칭 , 음수 좌측의 긴 꼬리(우편향), 양수 우측의 긴 꼬리 편향(좌편향)
print("왜도(치우친 정도) :",skew(data))
print("첨도(뾰족한 정도) :",kurtosis(data))

## 맥니마 검정 => 쿠폰을 뿌렸을 때 상태가 변하는 지 안 변하는 지
data = pd.DataFrame({'치료후음성' : [59,16,75],
                    '치료후양성' : [6,80,86]},
                   index = ['치료전양성','치료전음성','계'])
data['계'] = data['치료후음성'] + data['치료후양성']
table = np.array(data.iloc[:2, :2])
table

# 치료가 음성과 양성에 영향을 미치는 지 검정
from statsmodels.stats.contingency_tables import mcnemar
result = mcnemar(table,
                 exact = True, # True인 경우 이산형 데이터이거나 데이터가 작을 때, False인 경우 연속형이거나 데이터가 크거나
                 correction = True)
result.statistic, result.pvalue # (6.0, 0.052478790283203125)
